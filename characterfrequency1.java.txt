public class CharacterFrequency {
    public static void main(String[] args) {
        String str = "hello";
        for (int i = 0; i < str.length(); i++) {
            char ch = str.charAt(i);
            int count = 0;
            for (int j = 0; j < str.length(); j++) {
                if (ch == str.charAt(j)) {
                    count++;
                }
            }
            boolean alreadyPrinted = false;
            for (int k = 0; k < i; k++) {
                if (ch == str.charAt(k)) {
                    alreadyPrinted = true;
                    break;
                }
            }
            if (!alreadyPrinted) {
                System.out.println(ch + " = " + count);
            }
        }
    }
}
